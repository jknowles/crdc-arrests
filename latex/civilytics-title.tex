% Civilytics — title-page partial.
%
% Loaded via include-in-header (NOT include-before-body). Pandoc places
% include-before-body content *after* \maketitle, so using that hook would print
% Quarto's default title block and then this one underneath it. Redefining
% \maketitle in the preamble instead means Quarto's own \maketitle call renders
% this design and nothing is duplicated.
%
% Requires latex/civilytics.tex to be loaded first (fonts, colors, titling,
% etoolbox).

% \thesubtitle is not provided by `titling` and only exists when the YAML sets a
% subtitle. Provide an empty fallback so the \ifdefempty test below is safe on
% documents without one. (The original template tested \ifx\thesubtitle\@empty
% outside \makeatletter, which fails: @ is not a letter in the document body.)
\providecommand{\thesubtitle}{}

\renewcommand{\maketitle}{%
\begin{titlepage}
  \pagecolor{paper}
  \color{ink}
  \vspace*{0.5in}
  {\sffamily\bfseries\scriptsize\color{ember}\MakeUppercase{— Civilytics Research}\par}
  \vspace{12pt}
  % \raggedright: a display title at 32pt justified across the full measure
  % stretches interword space badly (the first render spaced "Using Small Area
  % Estimation to" almost to double). Display type is set ragged right.
  {\raggedright\displayfont\fontsize{32pt}{34pt}\selectfont\bfseries\color{ink}\thetitle\par}
  \vspace{8pt}
  \ifdefempty{\thesubtitle}{}{%
    {\rmfamily\itshape\fontsize{14pt}{18pt}\selectfont\color{ink2}\thesubtitle\par}
  }
  \vspace{18pt}
  \noindent\rule{\linewidth}{2pt}\par
  \vspace{1pt}
  \noindent{\color{rule}\rule{\linewidth}{0.4pt}}\par
  \vspace{8pt}
  \noindent
  \begin{minipage}[t]{0.5\linewidth}
    {\sffamily\scriptsize\color{ink3}\MakeUppercase{Authors}\par}
    \vspace{2pt}
    {\sffamily\small\color{ink}\theauthor\par}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.4\linewidth}\raggedleft
    {\sffamily\scriptsize\color{ink3}\MakeUppercase{Published}\par}
    \vspace{2pt}
    {\sffamily\small\color{ink}\thedate\par}
  \end{minipage}\par
  \vspace{1.5cm}

  % Pulse mark, in ember. The original wrapped this in an empty tikzpicture
  % without loading tikz, which is an undefined-environment error; the rule is
  % the only thing that ever drew anything, so the wrapper is gone.
  \begin{center}
    {\color{ember}\rule{40pt}{2pt}}
  \end{center}
\end{titlepage}
\clearpage
}
